````bash
# A3: every line has a status, a stage and a check, or is n/a with a reason
awk '/^```/{f=!f; next} f{next}
 /^## /{s=$2; c[s]=0; next} !s{next}
 /^- /{c[s]++; if ($0 ~ /^- n\/a: ./) next
   if ($0 !~ /^- \[[x ]\] /) print "no status: " $0
   else if ($0 ~ /^- \[x\] unchecked:/) print "unchecked marked passed: " $0
   else if ($0 !~ / · stage: (foundation|building|hardening|live)( |$)/) print "no stage: " $0
   else if ($0 !~ / · check: ./) print "no check: " $0
   next}
 /^n\/a: ./{c[s]++}
 END{for (k in c) if (!c[k]) print "empty section: " k}' CHECKLIST.md; echo "lines checked"

# A4: the old lines, by their claim (the text before " · check:")
git show origin/develop:BACKLOG.md | sed -n '/^## Repo checklist/,$p' | grep -E '^- ' \
 | sed -E 's/^- (\[.\] )?(unchecked: |n\/a: )?//; s/ · check:.*//' \
 | while IFS= read -r l; do grep -qF -- "$l" CHECKLIST.md || echo "missing: $l"; done; echo "old lines checked"

# A4: BACKLOG.md outside the old section ("Repo checklist" is its last section on develop):
# no "<" line (nothing removed or changed); the ">" lines are the pointer and this card's own Parked lines
diff -B <(git show origin/develop:BACKLOG.md | sed '/^## Repo checklist/,$d') BACKLOG.md | grep '^[<>]'

# A5: per-section counts
awk '/^```/{f=!f; next} f{next}
 /^## /{s=$2; o[++k]=s; next} !s{next}
 /^- \[x\]/{p[s]++; next} /^- \[ \] unchecked:/{u[s]++; next} /^- \[ \]/{x[s]++; next}
 /^(- )?n\/a:/{na[s]++}
 END{print "| section | pass | fail | unchecked | n/a |"; print "| --- | --- | --- | --- | --- |"
   for (i=1;i<=k;i++){t=o[i]; printf "| %s | %d | %d | %d | %d |\n", t, p[t], x[t], u[t], na[t]}}' CHECKLIST.md
````
